  <script>
    let heartScore = 0;

    // Generate Floating Background Balloons
    const balloonContainer = document.getElementById('balloon-container');
    const colors = ['#ff4b72', '#ff9a9e', '#fecfef', '#a1c4fd', '#ffd166'];
    
    for (let i = 0; i < 15; i++) {
      let balloon = document.createElement('div');
      balloon.className = 'balloon';
      balloon.style.left = Math.random() * 100 + 'vw';
      balloon.style.animationDelay = Math.random() * 8 + 's';
      balloon.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
      balloonContainer.appendChild(balloon);
    }

    // Navigation function
    function nextScreen(screenNumber) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      document.getElementById('screen' + screenNumber).classList.add('active');
    }

    // Screen 2 Game Logic
    function catchHeart() {
      heartScore++;
      document.getElementById('heart-count').innerText = `Caught: ${heartScore}/5`;
      
      const heart = document.getElementById('target-heart');
      const gameArea = document.getElementById('game-area');
      
      if (heartScore >= 5) {
        confetti({ particleCount: 50, spread: 60, origin: { y: 0.6 } });
        setTimeout(() => nextScreen(3), 500);
      } else {
        // Move heart to random position
        const maxX = gameArea.clientWidth - 40;
        const maxY = gameArea.clientHeight - 40;
        heart.style.left = Math.random() * maxX + 'px';
        heart.style.top = Math.random() * maxY + 'px';
      }
    }

    // Screen 3 Candle Logic
    function blowCandle() {
      document.getElementById('flame').classList.add('off');
      confetti({ particleCount: 80, spread: 70, origin: { y: 0.6 } });
      setTimeout(() => nextScreen(4), 1200);
    }

    // Screen 4 Gift Logic
    function openGift() {
      document.getElementById('gift-text').style.display = 'block';
      confetti({ particleCount: 100, spread: 100, origin: { y: 0.5 } });
      setTimeout(() => nextScreen(5), 2500);
    }

    // Reset All State
    function resetAll() {
      heartScore = 0;
      document.getElementById('heart-count').innerText = 'Caught: 0/5';
      document.getElementById('flame').classList.remove('off');
      document.getElementById('gift-text').style.display = 'none';
      const heart = document.getElementById('target-heart');
      heart.style.left = '45%';
      heart.style.top = '40%';
    }
  </script>
